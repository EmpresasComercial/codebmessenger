=== public.get_my_recharges_summary_mcpn ===
CREATE OR REPLACE FUNCTION public.get_my_recharges_summary_mcpn()
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
    v_user_id UUID := auth.uid();
    v_total_aprovado NUMERIC := 0;
    v_total_pendente NUMERIC := 0;
    v_qtd_total INT := 0;
    v_qtd_aprovadas INT := 0;
    v_qtd_pendentes INT := 0;
    v_recharges JSONB := '[]'::jsonb;
BEGIN
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Acesso não autorizado: autenticação obrigatória.';
    END IF;

    SELECT 
        COALESCE(SUM(CASE WHEN r.status IN ('aprovado', 'concluido', 'approved', 'completed') THEN r.valor ELSE 0 END), 0),
        COALESCE(SUM(CASE WHEN r.status IN ('pendente', 'pending', 'recarregando') THEN r.valor ELSE 0 END), 0),
        COUNT(*),
        COALESCE(COUNT(*) FILTER (WHERE r.status IN ('aprovado', 'concluido', 'approved', 'completed')), 0),
        COALESCE(COUNT(*) FILTER (WHERE r.status IN ('pendente', 'pending', 'recarregando')), 0)
    INTO 
        v_total_aprovado,
        v_total_pendente,
        v_qtd_total,
        v_qtd_aprovadas,
        v_qtd_pendentes
    FROM public.paymentbot r
    WHERE r.user_id = v_user_id;

    SELECT COALESCE(
        jsonb_agg(
            jsonb_build_object(
                'id', r.id,
                'valor', r.valor,
                'amount', r.valor,
                'status', r.status,
                'created_at', r.created_at,
                'banco_origem', COALESCE(
                    r.detalhes_transacao->>'banco_origem',
                    r.detalhes_transacao->>'banco_destino',
                    r.detalhes_transacao->>'banco',
                    'Depósito Bancário'
                ),
                'chave_transacao', COALESCE(r.detalhes_transacao->>'chave_transacao', ''),
                'comprovativo_url', COALESCE(r.detalhes_transacao->>'comprovativo_url', ''),
                'detalhes_transacao', r.detalhes_transacao
            ) ORDER BY r.created_at DESC
        ), '[]'::jsonb
    )
    INTO v_recharges
    FROM public.paymentbot r
    WHERE r.user_id = v_user_id;

    RETURN jsonb_build_object(
        'success', true,
        'total_aprovado', v_total_aprovado,
        'total_pendente', v_total_pendente,
        'qtd_total', v_qtd_total,
        'qtd_aprovadas', v_qtd_aprovadas,
        'qtd_pendentes', v_qtd_pendentes,
        'recharges', v_recharges
    );
END;
$function$

=== public.getwallet ===
CREATE OR REPLACE FUNCTION public.getwallet()
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'private_mcpn', 'pg_temp'
AS $function$
DECLARE
    v_user_id UUID := auth.uid();
    v_user_balance NUMERIC := 0;
    v_total_retirado NUMERIC := 0;
    v_has_pending BOOLEAN := false;
    v_has_bank BOOLEAN := false;
    v_bank_name TEXT := NULL;
    v_iban TEXT := NULL;
    v_has_bots BOOLEAN := false;
    v_now_luanda TIMESTAMP := (now() AT TIME ZONE 'Africa/Luanda');
    v_day_of_week INT := EXTRACT(DOW FROM v_now_luanda);
    v_hour INT := EXTRACT(HOUR FROM v_now_luanda);
    v_is_banking_hours BOOLEAN := false;
    v_payload JSONB;
    v_encrypted_token TEXT;
BEGIN
    -- Validação estrita de autenticação
    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Acesso não autorizado: autenticação obrigatória.';
    END IF;

    -- Horário bancário: Segunda a Sexta (1 a 5) das 09:00 às 18:00
    IF v_day_of_week BETWEEN 1 AND 5 AND v_hour >= 9 AND v_hour < 18 THEN
        v_is_banking_hours := true;
    END IF;

    -- Saldo e total retirado de nucleo
    SELECT COALESCE(saldo_disponivel, 0), COALESCE(total_retirado, 0)
    INTO v_user_balance, v_total_retirado
    FROM public.nucleo
    WHERE id = v_user_id;

    -- Retirada pendente
    SELECT EXISTS(
        SELECT 1 FROM public.retiradas_mcpn
        WHERE user_id = v_user_id AND status = 'pendente'
    ) INTO v_has_pending;

    -- Conta bancária vinculada em sys_t111
    SELECT 
        true,
        (dados_bancarios->>'nome_banco')::text,
        COALESCE(private_mcpn.decrypt_data_mcpn(dados_bancarios->>'iban'), 'IBAN Registrado')
    INTO v_has_bank, v_bank_name, v_iban
    FROM public.sys_t111
    WHERE user_id = v_user_id
    LIMIT 1;

    -- Bots ativos
    SELECT EXISTS(
        SELECT 1 FROM public.sys_600
        WHERE user_id = v_user_id AND ativo = true AND (data_fim IS NULL OR data_fim > now())
    ) INTO v_has_bots;

    -- Montagem do payload oficial com taxa 14% (fator líquido 0.86)
    v_payload := jsonb_build_object(
        'bot_type', 'wallet_assistant',
        'currency', 'KZ',
        'min_amount', 3000,
        'max_amount', 500000,
        'fee_percent', 14,
        'net_factor', 0.86,
        'formula', 'valor_liquido = valor_bruto * 0.86',
        'schedule', 'Segunda a Sexta-feira, das 09:00 às 18:00 (dias úteis)',
        'is_banking_hours_now', v_is_banking_hours,
        'processing_time', 'De instantes a poucas horas em horário bancário (máx. 24h úteis)',
        'user', jsonb_build_object(
            'authenticated', true,
            'balance', v_user_balance,
            'total_withdrawn', v_total_retirado,
            'has_pending', v_has_pending,
            'has_bank', COALESCE(v_has_bank, false),
            'bank_name', v_bank_name,
            'has_bots', v_has_bots
        ),
        'timestamp', EXTRACT(EPOCH FROM now())
    );

    BEGIN
        v_encrypted_token := private_mcpn.encrypt_data_mcpn(v_payload::text);
    EXCEPTION WHEN OTHERS THEN
        v_encrypted_token := encode(v_payload::text::bytea, 'base64');
    END;

    RETURN jsonb_build_object(
        'success', true,
        'min_amount', 3000,
        'max_amount', 500000,
        'fee_percent', 14,
        'net_factor', 0.86,
        'formula', 'valor_liquido = valor_bruto * 0.86',
        'schedule', 'Segunda a Sexta-feira, das 09:00 às 18:00 (dias úteis)',
        'is_banking_hours_now', v_is_banking_hours,
        'processing_time', 'De instantes a poucas horas em horário bancário (máx. 24h úteis)',
        'user_balance', v_user_balance,
        'has_pending', v_has_pending,
        'has_bank', COALESCE(v_has_bank, false),
        'bank_name', v_bank_name,
        'has_bots', v_has_bots,
        'encrypted_token', v_encrypted_token
    );
END;
$function$

=== private_mcpn.distribute_purchase_commission ===
CREATE OR REPLACE FUNCTION private_mcpn.distribute_purchase_commission(p_buyer_id uuid, p_amount numeric)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'extensions', 'private_mcpn'
AS $function$
DECLARE
    v_l1_id UUID; v_l2_id UUID; v_l3_id UUID;
    v_is_first_purchase BOOLEAN;
BEGIN
    IF p_amount < 10000 THEN RETURN; END IF;

    SELECT (COUNT(*) = 1) INTO v_is_first_purchase
    FROM public.sys_600
    WHERE user_id = p_buyer_id;

    IF v_is_first_purchase THEN
        -- Nível 1 (10%)
        SELECT patrocinador_id INTO v_l1_id FROM public.equipe_mcpn WHERE usuario_id = p_buyer_id;
        IF v_l1_id IS NOT NULL THEN
            UPDATE public.nucleo SET
                saldo_disponivel = saldo_disponivel + (p_amount * 0.10)
            WHERE id = v_l1_id;
            INSERT INTO public.extrato_mcpn (user_id, tipo, from_user_id, valor, nivel, detalhes)
            VALUES (v_l1_id, 'comissao_equipe', p_buyer_id, (p_amount * 0.10), 1,
                jsonb_build_object('descricao', 'Comissão Compra Direta', 'porcentagem_aplicada', 10))
            ON CONFLICT ON CONSTRAINT unique_comissao_evento DO NOTHING;

            -- Nível 2 (6%)
            SELECT patrocinador_id INTO v_l2_id FROM public.equipe_mcpn WHERE usuario_id = v_l1_id;
            IF v_l2_id IS NOT NULL THEN
                UPDATE public.nucleo SET
                    saldo_disponivel = saldo_disponivel + (p_amount * 0.06)
                WHERE id = v_l2_id;
                INSERT INTO public.extrato_mcpn (user_id, tipo, from_user_id, valor, nivel, detalhes)
                VALUES (v_l2_id, 'comissao_equipe', p_buyer_id, (p_amount * 0.06), 2,
                    jsonb_build_object('descricao', 'Comissão Compra Indireta L2', 'porcentagem_aplicada', 6))
                ON CONFLICT ON CONSTRAINT unique_comissao_evento DO NOTHING;

                -- Nível 3 (2%)
                SELECT patrocinador_id INTO v_l3_id FROM public.equipe_mcpn WHERE usuario_id = v_l2_id;
                IF v_l3_id IS NOT NULL THEN
                    UPDATE public.nucleo SET
                        saldo_disponivel = saldo_disponivel + (p_amount * 0.02)
                    WHERE id = v_l3_id;
                    INSERT INTO public.extrato_mcpn (user_id, tipo, from_user_id, valor, nivel, detalhes)
                    VALUES (v_l3_id, 'comissao_equipe', p_buyer_id, (p_amount * 0.02), 3,
                        jsonb_build_object('descricao', 'Comissão Compra Indireta L3', 'porcentagem_aplicada', 2))
                    ON CONFLICT ON CONSTRAINT unique_comissao_evento DO NOTHING;
                END IF;
            END IF;
        END IF;
    END IF;
END;

CREATE OR REPLACE FUNCTION private_mcpn.collect_daily_earnings()
 RETURNS json
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public', 'extensions', 'private_mcpn'
AS $function$
DECLARE
    v_user_id UUID := auth.uid();
    v_total_collected NUMERIC := 0;
    v_already_collected BOOLEAN;
    v_today_luanda DATE := (now() AT TIME ZONE 'Africa/Luanda')::date;
BEGIN
    IF v_user_id IS NULL THEN
        RETURN json_build_object('success', false, 'message', 'Sessão expirada.');
    END IF;

    -- Verificar se já coletou hoje (usar extrato_mcpn)
    SELECT EXISTS (
        SELECT 1 FROM public.extrato_mcpn
        WHERE user_id = v_user_id
          AND tipo = 'renda_diaria'
          AND (created_at AT TIME ZONE 'Africa/Luanda')::date = v_today_luanda
    ) INTO v_already_collected;

    IF v_already_collected THEN
        RETURN json_build_object('success', false, 'message', 'Você já coletou seus rendimentos hoje.');
    END IF;

    SELECT COALESCE(SUM(renda_diaria), 0) INTO v_total_collected
    FROM public.sys_600
    WHERE user_id = v_user_id AND ativo = true AND (dias_restantes > 0 OR dias_restantes IS NULL);

    IF v_total_collected <= 0 THEN
        RETURN json_build_object('success', false, 'message', 'Sem rendimentos ativos.');
    END IF;

    UPDATE public.sys_600 SET
        dias_restantes = CASE WHEN dias_restantes IS NOT NULL THEN dias_restantes - 1 ELSE NULL END,
        ativo = CASE WHEN dias_restantes IS NOT NULL AND (dias_restantes - 1) <= 0 THEN false ELSE true END
    WHERE user_id = v_user_id AND ativo = true;

    UPDATE public.nucleo SET
        saldo_disponivel = saldo_disponivel + v_total_collected,
        updated_at = now()
    WHERE id = v_user_id;

    -- Inserir no extrato unificado
    INSERT INTO public.extrato_mcpn (user_id, tipo, valor)
    VALUES (v_user_id, 'renda_diaria', v_total_collected);

    RETURN json_build_object('success', true, 'total_collected', v_total_collected);
END;
$function$
